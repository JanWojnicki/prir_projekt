cmake_minimum_required(VERSION 3.24)
project(C05Entropy VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(C05_ENABLE_OPENMP "Build OpenMP backend" ON)
option(C05_ENABLE_MPI "Build MPI backend" ON)
option(C05_ENABLE_CUDA "Build CUDA backend" ON)
option(C05_REQUIRE_ALL "Fail if any requested backend is unavailable" OFF)
option(C05_NATIVE "Enable host -march=native" ON)
add_library(c05_flags INTERFACE)
target_compile_options(c05_flags INTERFACE
  $<$<COMPILE_LANG_AND_ID:CXX,GNU,Clang>:-O3;-Wall;-Wextra>
  $<$<COMPILE_LANG_AND_ID:CUDA,NVIDIA>:-O3;-Xcompiler=-Wall,-Wextra>)
if(C05_NATIVE)
  target_compile_options(c05_flags INTERFACE
    $<$<COMPILE_LANG_AND_ID:CXX,GNU,Clang>:-march=native>
    $<$<COMPILE_LANG_AND_ID:CUDA,NVIDIA>:-Xcompiler=-march=native>)
endif()
add_library(common STATIC common/io.cpp common/compute.cpp)
target_include_directories(common PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(common PUBLIC c05_flags)
add_executable(entropy_ref src/reference.cpp)
target_link_libraries(entropy_ref PRIVATE common)
if(C05_ENABLE_OPENMP)
  find_package(OpenMP QUIET COMPONENTS CXX)
  if(OpenMP_CXX_FOUND)
    add_executable(entropy_omp src/openmp.cpp)
    target_link_libraries(entropy_omp PRIVATE common OpenMP::OpenMP_CXX)
  elseif(C05_REQUIRE_ALL)
    message(FATAL_ERROR "OpenMP CXX is required")
  else()
    message(STATUS "OpenMP unavailable: entropy_omp not built")
  endif()
endif()
if(C05_ENABLE_MPI)
  find_package(MPI QUIET COMPONENTS CXX)
  if(MPI_CXX_FOUND)
    add_executable(entropy_mpi src/mpi.cpp)
    target_link_libraries(entropy_mpi PRIVATE common MPI::MPI_CXX)
  elseif(C05_REQUIRE_ALL)
    message(FATAL_ERROR "MPI CXX is required")
  else()
    message(STATUS "MPI unavailable: entropy_mpi not built")
  endif()
endif()
if(C05_ENABLE_CUDA)
  include(CheckLanguage)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "CUDA architectures, e.g. 75 or native")
  endif()
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    set(CMAKE_CUDA_STANDARD 20)
    set(CMAKE_CUDA_STANDARD_REQUIRED ON)
    add_executable(entropy_cuda src/cuda_entropy.cu)
    target_link_libraries(entropy_cuda PRIVATE common CUDA::cudart)
    target_compile_options(entropy_cuda PRIVATE $<$<COMPILE_LANG_AND_ID:CUDA,NVIDIA>:-lineinfo>)
    set_target_properties(entropy_cuda PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
  elseif(C05_REQUIRE_ALL)
    message(FATAL_ERROR "CUDA compiler is required")
  else()
    message(STATUS "CUDA unavailable: entropy_cuda not built")
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME reference_correctness COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/verify_correctness.py
      --exe $<TARGET_FILE:entropy_ref>)
  endif()
endif()

if(BUILD_TESTING AND Python3_Interpreter_FOUND AND TARGET entropy_cuda)
  add_test(NAME cuda_correctness COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/verify_cuda.py
    --ref-exe $<TARGET_FILE:entropy_ref> --cuda-exe $<TARGET_FILE:entropy_cuda>)
  set_tests_properties(cuda_correctness PROPERTIES LABELS "gpu")
endif()
